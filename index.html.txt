function n(id){ return parseFloat(document.getElementById(id)?.value)||0; }
function fmt(x){ return '₹'+Math.abs(Math.round(x)).toLocaleString('en-IN'); }

// ══════════════════════════════════════
// TAX CALCULATION
// ══════════════════════════════════════
function calcNewRegimeTax(income){
  const slabs = [
    {limit:400000, rate:0},
    {limit:800000, rate:0.05},
    {limit:1200000, rate:0.10},
    {limit:1600000, rate:0.15},
    {limit:2000000, rate:0.20},
    {limit:2400000, rate:0.25},
    {limit:Infinity, rate:0.30}
  ];
  let tax=0, prev=0, breakdown=[];
  for(const s of slabs){
    if(income<=prev) break;
    const taxable=Math.min(income,s.limit)-prev;
    const t=taxable*s.rate;
    if(taxable>0) breakdown.push({from:prev,to:Math.min(income,s.limit),rate:s.rate*100,tax:t});
    tax+=t; prev=s.limit;
  }
  // Rebate 87A
  let rebate=0;
  if(income<=1200000) rebate=tax;
  tax=Math.max(0,tax-rebate);
  const cess=tax*0.04;
  return {tax,cess,total:tax+cess,rebate,breakdown};
}

function calcOldRegimeTax(income, category){
  let exempt=250000;
  if(category==='senior') exempt=300000;
  if(category==='supersenior') exempt=500000;
  const slabs=[
    {limit:exempt, rate:0},
    {limit:500000, rate:0.05},
    {limit:1000000, rate:0.20},
    {limit:Infinity, rate:0.30}
  ];
  let tax=0, prev=0, breakdown=[];
  for(const s of slabs){
    if(income<=prev) break;
    const taxable=Math.min(income,s.limit)-prev;
    const t=taxable*s.rate;
    if(taxable>0) breakdown.push({from:prev,to:Math.min(income,s.limit),rate:s.rate*100,tax:t});
    tax+=t; prev=s.limit;
  }
  // Rebate 87A – old regime
  let rebate=0;
  if(income<=500000) rebate=Math.min(tax,12500);
  tax=Math.max(0,tax-rebate);
  const cess=tax*0.04;
  return {tax,cess,total:tax+cess,rebate,breakdown};
}

function getSurcharge(income, tax){
  if(income<=5000000) return 0;
  if(income<=10000000) return tax*0.10;
  if(income<=20000000) return tax*0.15;
  if(income<=50000000) return tax*0.25;
  return tax*0.37;
}

// MAIN RECALCULATE

function recalculate(){
  const gross = n('i-gross');
  const exempt = n('i-exempt');
  const epf = n('i-epf');
  const house = n('i-house');
  const biz = n('i-biz');
  const stcg = n('i-stcg');
  const ltcg = n('i-ltcg');
  const interest = n('i-interest');
  const dividend = n('i-dividend');
  const other = n('i-other');

  // Standard Deduction
  const stdDed = state.regime==='new' ? 75000 : (n('d-std')||50000);

  // Salary after exemptions
  const salaryAfterExempt = Math.max(0, gross - exempt - epf);
  const salaryAfterStd = Math.max(0, salaryAfterExempt - stdDed);

  // Gross Total Income
  const gti = salaryAfterStd + house + biz + stcg + ltcg + interest + dividend + other;

  // Deductions (only old regime)
  let totalDed = 0;
  if(state.regime==='old'){
    const d80c = Math.min(n('d-80c'), 150000);
    const dNPS = Math.min(n('d-nps'), 50000);
    const d80d = n('d-80d');
    const d80e = n('d-80e');
    const d80eea = Math.min(n('d-80eea'), 150000);
    const d80g = n('d-80g');
    const d80tta = Math.min(n('d-80tta'), 10000);
    const d80ttb = Math.min(n('d-80ttb'), 50000);
    const dHL = Math.min(n('d-homeloan'), 200000);
    const dPT = n('d-pt');
    const dHRA = n('d-hra');
    totalDed = d80c+dNPS+d80d+d80e+d80eea+d80g+d80tta+d80ttb+dHL+dPT+dHRA;

    // Update deductions summary
    const items = [
      ['80C (LIC/PPF/ELSS/PF)', d80c],
      ['80CCD(1B) NPS', dNPS],
      ['80D Health Insurance', d80d],
      ['80E Education Loan', d80e],
      ['80EEA Housing (Affordable)', d80eea],
      ['80G Donations', d80g],
      ['80TTA Savings Interest', d80tta],
      ['80TTB Interest (Senior)', d80ttb],
      ['24(b) Home Loan Interest', dHL],
      ['Professional Tax', dPT],
      ['HRA Exemption', dHRA],
    ];
    document.getElementById('deductions-summary').innerHTML = items.filter(i=>i[1]>0).map(i=>`
      <div class="summary-line">
        <span class="summary-key">${i[0]}</span>
        <span class="summary-val deduction">${fmt(i[1])}</span>
      </div>`).join('') || '<div class="summary-line"><span class="summary-key" style="color:var(--muted)">No deductions entered</span></div>';
  } else {
    document.getElementById('deductions-summary').innerHTML = '<div style="font-size:13px;color:var(--muted);padding:8px 0">Deductions u/s 80C, 80D, etc. are not available under New Regime (except NPS employer 80CCD(2)).</div>';
  }

  document.getElementById('s-deductions').textContent = fmt(totalDed);

  const nti = Math.max(0, gti - totalDed);

  // Calc both regimes
  const category = document.getElementById('p-category')?.value || 'individual';
  const newTax = calcNewRegimeTax(Math.max(0, salaryAfterStd + house + biz + stcg + ltcg + interest + dividend + other));
  const oldTax = calcOldRegimeTax(nti, category);

  // Surcharge
  const nSurcharge = getSurcharge(nti, newTax.tax);
  const oSurcharge = getSurcharge(nti, oldTax.tax);
  const newTotal = newTax.total + nSurcharge;
  const oldTotal = oldTax.total + oSurcharge;

  // Current regime
  const curResult = state.regime==='new' ? newTax : oldTax;
  const curTotal = state.regime==='new' ? newTotal : oldTotal;

  // Update income panel
  document.getElementById('s-gti').textContent = fmt(gti);
  document.getElementById('s-nti').textContent = fmt(nti);
  document.getElementById('s-tax').textContent = fmt(curTotal);

  // TDS & advance tax
  const tdsTot = state.tdsEntries.reduce((a,b)=>a+b.tds,0);
  const atTot = state.advTaxEntries.reduce((a,b)=>a+b.amt,0);
  const credits = tdsTot + atTot;
  document.getElementById('s-tds-total').textContent = fmt(tdsTot);
  document.getElementById('s-at-total').textContent = fmt(atTot);
  const sat = curTotal - credits;
  document.getElementById('s-sat').textContent = fmtSign(sat);
  document.getElementById('s-sat').style.color = sat<0?'var(--green)':'var(--accent)';

  // ── SUMMARY TAB ──
  // Comparison
  const newWinner = newTotal <= oldTotal;
  document.getElementById('cmp-new-val').textContent = fmt(newTotal);
  document.getElementById('cmp-old-val').textContent = fmt(oldTotal);
  document.getElementById('cmp-new').classList.toggle('winner', newWinner);
  document.getElementById('cmp-old').classList.toggle('winner', !newWinner);
  document.getElementById('cmp-recommend').textContent = newWinner ? 'New Regime' : 'Old Regime';
  const savings = Math.abs(newTotal - oldTotal);
  document.getElementById('cmp-savings').textContent = savings>0
    ? `💡 ${newWinner?'New':'Old'} Regime saves you ${fmt(savings)} in tax.`
    : 'Both regimes result in equal tax.';

  // Income computation
  document.getElementById('income-comp').innerHTML = `
    <div class="summary-line"><span class="summary-key">Gross Salary</span><span class="summary-val">${fmt(gross)}</span></div>
    <div class="summary-line"><span class="summary-key">Less: Exemptions (HRA/LTA)</span><span class="summary-val deduction">${fmt(exempt)}</span></div>

    <div class="summary-line"><span class="summary-key">STCG</span><span class="summary-val">${fmt(stcg)}</span></div>
    <div class="summary-line"><span class="summary-key">LTCG</span><span class="summary-val">${fmt(ltcg)}</span></div>
    <div class="summary-line"><span class="summary-key">Interest / Dividend / Other</span><span class="summary-val">${fmt(interest+dividend+other)}</span></div>
    <div class="summary-line"><span class="summary-key bold">Gross Total Income</span><span class="summary-val">${fmt(gti)}</span></div>
    ${state.regime==='old'?`<div class="summary-line"><span class="summary-key">Less: Chapter VI-A Deductions</span><span class="summary-val deduction">${fmt(totalDed)}</span></div>`:''}
    <div class="summary-line total"><span class="summary-key bold">Net Taxable Income</span><span>${fmt(nti)}</span></div>
  `;

  // Tax computation
  document.getElementById('tax-comp').innerHTML = `
    <div class="summary-line"><span class="summary-key">Tax on Normal Income</span><span class="summary-val tax">${fmt(curResult.tax)}</span></div>
    <div class="summary-line"><span class="summary-key">Less: Rebate u/s 87A</span><span class="summary-val deduction">${fmt(curResult.rebate)}</span></div>
    <div class="summary-line"><span class="summary-key">Surcharge</span><span class="summary-val">${fmt(state.regime==='new'?nSurcharge:oSurcharge)}</span></div>
    <div class="summary-line"><span class="summary-key">Health & Education Cess (4%)</span><span class="summary-val">${fmt(curResult.cess)}</span></div>
    <div class="summary-line total"><span class="summary-key bold">Total Tax Liability</span><span>${fmt(curTotal)}</span></div>
  `;

  // Final
  document.getElementById('f-gross-tax').textContent = fmt(curTotal);
  document.getElementById('f-credits').textContent = fmt(credits);
  const bal = curTotal - credits;
  document.getElementById('f-result').textContent = fmt(Math.abs(bal));
  document.getElementById('f-result-label').textContent = bal>=0 ? 'Net Tax Payable' : 'Refund Due';
  document.getElementById('f-result-note').textContent = bal>=0 ? 'Pay via Challan 280' : 'File ITR to claim refund';
  document.getElementById('f-result-box').className = 'stat-box '+(bal>=0?'amber':'green');

  // Slab breakdown
  const slabs = state.regime==='new' ? newTax.breakdown : oldTax.breakdown;
  document.getElementById('slab-breakdown').innerHTML = `
    <div class="tbl-wrap">
    <table>
      <thead><tr>
        <th>Income Slab</th>
        <th>Rate</th>
        <th>Taxable Amount</th>
        <th>Tax</th>
        <th>%</th>
      </tr></thead>
      <tbody>
        ${slabs.map(s=>`<tr>
          <td>${fmt(s.from)} – ${s.to===Infinity?'Above':fmt(s.to)}</td>
          <td><span class="badge ${s.rate===0?'badge-green':s.rate<=15?'badge-blue':s.rate<=25?'badge-amber':'badge-red'}">${s.rate}%</span></td>
          <td class="td-num">${fmt(s.to===Infinity?nti-s.from:Math.min(nti,s.to)-s.from)}</td>
          <td class="td-num">${fmt(s.tax)}</td>
          <td>
            <div class="progress-bar"><div class="progress-fill" style="width:${curTotal>0?Math.min(100,s.tax/curTotal*100):0}%;background:${s.rate===0?'var(--green)':s.rate<=15?'var(--blue)':s.rate<=25?'var(--accent)':'var(--accent2)'}"></div></div>
          </td>
        </tr>`).join('')}
      </tbody>
    </table>
    </div>`;
}

function exportPDF(){
  const { jsPDF } = window.jspdf;
  const doc = new jsPDF({unit:'mm',format:'a4'});
  const W=210, M=15;
  let y=M;

  // Header
  doc.setFillColor(13,15,20);
  doc.rect(0,0,W,28,'F');
  doc.setTextColor(245,166,35);
  doc.setFontSize(16);doc.setFont('helvetica','bold');

  doc.text(`Generated: ${new Date().toLocaleDateString('en-IN',{day:'2-digit',month:'long',year:'numeric'})}`,W-M,20,{align:'right'});

  const pan=document.getElementById('p-pan').value||'—';
  const category=document.getElementById('p-category').value||'individual';
  doc.setFontSize(11);doc.setFont('helvetica','bold');doc.setTextColor(30,36,51);
  doc.setFillColor(240,242,248);doc.rect(M,y,W-2*M,8,'F');
  doc.setTextColor(50,50,50);doc.text('TAXPAYER DETAILS',M+3,y+5.5);
  y+=12;
  doc.setFont('helvetica','normal');doc.setFontSize(9);
  doc.setTextColor(80,80,80);
  doc.text(`Name: ${name}`,M,y);doc.text(`PAN: ${pan}`,M+80,y);
  doc.text(`Category: ${category}`,M+140,y);
  y+=16;

  // Income
  const gross=n('i-gross'),exempt=n('i-exempt'),epf=n('i-epf');
  const stdDed=state.regime==='new'?75000:(n('d-std')||50000);
  const salaryNet=Math.max(0,gross-exempt-epf-stdDed);
  const house=n('i-house'),biz=n('i-biz'),stcg=n('i-stcg'),ltcg=n('i-ltcg');
  const interest=n('i-interest'),dividend=n('i-dividend'),other=n('i-other');
  const gti=salaryNet+house+biz+stcg+ltcg+interest+dividend+other;

  doc.autoTable({
    startY:y,
    head:[['Income Head','Amount (₹)']],
    body:[
      ['Gross Salary',fmt(gross)],
      ['Less: Exemptions (HRA/LTA)',fmt(exempt)],
      [`Less: Standard Deduction (${state.regime==='new'?'New':'Old'} Regime)`,fmt(stdDed)],
      ['Net Salary Income',fmt(salaryNet)],
      ['House Property Income / (Loss)',fmtSign(house)],
      ['Business / Professional Income',fmt(biz)],
      ['Short Term Capital Gains',fmt(stcg)],
      ['Long Term Capital Gains',fmt(ltcg)],
      ['Interest Income',fmt(interest)],
      ['Dividend Income',fmt(dividend)],
      ['Other Income',fmt(other)],
      ['GROSS TOTAL INCOME',fmt(gti)],
    ],
    theme:'grid',
    headStyles:{fillColor:[13,15,20],textColor:[245,166,35],fontStyle:'bold',fontSize:9},
    styles:{fontSize:9,cellPadding:3},
    columnStyles:{1:{halign:'right'}},
    margin:{left:M,right:M},
  });
  y=doc.lastAutoTable.finalY+8;

  // Deductions
  if(state.regime==='old'){
    const d80c=Math.min(n('d-80c'),150000),dNPS=Math.min(n('d-nps'),50000),d80d=n('d-80d');
    const d80e=n('d-80e'),d80eea=Math.min(n('d-80eea'),150000),d80g=n('d-80g');

    const totalDed=d80c+dNPS+d80d+d80e+d80eea+d80g+d80tta+d80ttb+dHL+dPT+dHRA;

    doc.autoTable({
      startY:y,
      head:[['Deductions (Chapter VI-A)','Amount (₹)']],
      body:[

  const taxRes=state.regime==='new'?calcNewRegimeTax(gti):calcOldRegimeTax(ntiForTax,category);

  const totalTax=taxRes.total+surcharge;
  const tdsTot=state.tdsEntries.reduce((a,b)=>a+b.tds,0);
  const atTot=state.advTaxEntries.reduce((a,b)=>a+b.amt,0);

      ['Less: TDS Deducted',fmt(tdsTot)],
      ['Less: Advance Tax Paid',fmt(atTot)],
      [bal>=0?'NET TAX PAYABLE':'REFUND DUE',fmtSign(Math.abs(bal))],

    theme:'grid',
    headStyles:{fillColor:[30,20,50],textColor:[245,166,35],fontStyle:'bold',fontSize:9},
    styles:{fontSize:9,cellPadding:3},
    columnStyles:{1:{halign:'right'}},
    margin:{left:M,right:M},
  });
  y=doc.lastAutoTable.finalY+8;

  // Assets
  if(state.assets.length){
    if(y>230){doc.addPage();y=M;}

      footStyles:{fillColor:[0,60,30],textColor:[255,255,255],fontStyle:'bold'},
      styles:{fontSize:9,cellPadding:3},
      columnStyles:{3:{halign:'right'}},
      margin:{left:M,right:M},
    });
    y=doc.lastAutoTable.finalY+8;
  }

  // Liabilities
  if(state.liabilities.length){
    if(y>230){doc.addPage();y=M;}
    doc.autoTable({
      startY:y,
      head:[['Liability Description','Category','Lender','Outstanding (₹)']],

      headStyles:{fillColor:[80,20,20],textColor:[255,255,255],fontStyle:'bold',fontSize:9},

      margin:{left:M,right:M},
    });

  }

  // Footer
  const pages=doc.internal.getNumberOfPages();
  for(let i=1;i<=pages;i++){
    doc.setPage(i);
    doc.setFontSize(8);doc.setTextColor(150,150,150);
    doc.text(`Page ${i} of ${pages} | Income Tax Assistant – CA Office Pro | For reference only. Consult a CA before filing.`,W/2,292,{align:'center'});
  }

  doc.save(`IncomeTax_${name.replace(/\s/g,'_')}_FY2025-26.pdf`);
  showToast('PDF downloaded ✓');
}
